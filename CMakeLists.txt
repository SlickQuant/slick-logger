cmake_minimum_required(VERSION 3.20)

project(slick-logger
    VERSION 2.0.0
    LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

option(SLICK_LOGGER_ENABLE_SOURCE_LOCATION "Enable LOG_* macro source-location capture" ON)
# Off drops the per-call capture and leaves entries carrying thread id 0. The
# LogEntry field is reserved either way, so this never changes the layout that
# shared-memory peers agree on - mixing the two settings across a segment is safe.
option(SLICK_LOGGER_ENABLE_THREAD_ID "Capture the producing thread id in every entry" ON)
# The multi-process collector is a convenience executable;
# the shared-memory feature itself is header-only and always available.
option(BUILD_SLICK_LOGGER_COLLECTOR "Build the standalone slick_log_collector executable" OFF)
option(BUILD_SLICK_LOGGER_BENCHMARKS "Build performance benchmarks" OFF)

# 2.1.0 or newer is required: the string ring is built with the items_per_slot
# constructor introduced in slick-queue 2.1.0.
find_package(slick-queue 2.1.0 CONFIG QUIET)

if (NOT slick-queue_FOUND)
    message(STATUS "Fetching slick-queue...")
    include(FetchContent)

    # Disable tests for slick_queue
    set(BUILD_SLICK_QUEUE_TESTS OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(
        slick-queue
        GIT_REPOSITORY https://github.com/SlickQuant/slick-queue.git
        GIT_TAG v2.1.0
    )
    FetchContent_MakeAvailable(slick-queue)
else()
    message(STATUS "Found slick-queue ${slick-queue_VERSION}: ${slick-queue_DIR}")
endif()

set(WARNING_MESSAGE "// ********** THIS FILE IS AUTO-GENERATED FROM src/logger.hpp **********
// ********** DO NOT EDIT THIS FILE DIRECTLY                  **********
// ********** EDIT src/logger.hpp AND RE-RUN CMAKE TO UPDATE  **********")
configure_file(
    src/logger.hpp
    ${CMAKE_CURRENT_SOURCE_DIR}/include/slick/logger.hpp
    NEWLINE_STYLE LF
)

if (NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()

if (CMAKE_BUILD_TYPE MATCHES Debug)
    add_definitions(-DDEBUG)
endif()

if (CMAKE_BUILD_TYPE MATCHES Release)
    add_definitions(-DNDEBUG)
endif()

# Add header-only library
add_library(slick-logger INTERFACE)
add_library(slick::logger ALIAS slick-logger)
# For backward compatibility with older versions
add_library(slick_logger ALIAS slick-logger)
add_library(slick::slick_logger ALIAS slick-logger)

target_include_directories(slick-logger INTERFACE
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:include>
)
set_target_properties(slick-logger PROPERTIES EXPORT_NAME logger)
if(NOT SLICK_LOGGER_ENABLE_SOURCE_LOCATION)
    target_compile_definitions(slick-logger INTERFACE SLICK_LOGGER_ENABLE_SOURCE_LOCATION=0)
endif()
if(NOT SLICK_LOGGER_ENABLE_THREAD_ID)
    target_compile_definitions(slick-logger INTERFACE SLICK_LOGGER_ENABLE_THREAD_ID=0)
endif()
target_link_libraries(slick-logger INTERFACE slick::queue)

if (MSVC)
    set(CMAKE_CXX_FLAGS_RELEASE "-O2")
    add_definitions(-D_WIN32_WINNT=0x0A00)
    set(CMAKE_SUPPRESS_REGENERATION true)   # supress zero_check
else()
    set(CMAKE_CXX_FLAGS_DEBUG "-g -O0 -fsanitize=address,undefined -fno-omit-frame-pointer")
    set(CMAKE_EXE_LINKER_FLAGS_DEBUG "-fsanitize=address,undefined")

    if (CMAKE_CXX_COMPILER_ID STREQUAL "Clang" AND NOT APPLE)
        # -flto=auto is a GCC spelling, and Clang's LTO emits LLVM bitcode that GNU
        # ld can only read through a gold plugin built from a matching LLVM version.
        # Distributions routinely ship a plugin older than the compiler (clang 18
        # against an LLVM 16 plugin, say), and the link then fails with
        # "LLVM gold plugin has failed to create LTO module: Unknown attribute kind".
        # LLD always matches its compiler, so use ThinLTO through LLD where it
        # exists and simply skip LTO where it does not: a slightly less optimized
        # Release build beats one that cannot link.
        find_program(SLICK_LOGGER_LLD NAMES ld.lld lld)
        if (SLICK_LOGGER_LLD)
            set(CMAKE_CXX_FLAGS_RELEASE "-O3 -march=native -flto=thin")
            set(CMAKE_EXE_LINKER_FLAGS_RELEASE "-flto=thin -fuse-ld=lld")
            message(STATUS "slick-logger: Clang ThinLTO via ${SLICK_LOGGER_LLD}")
        else()
            set(CMAKE_CXX_FLAGS_RELEASE "-O3 -march=native")
            message(STATUS "slick-logger: LLD not found, building Release without LTO")
        endif()
    else()
        # GCC, and AppleClang whose ld64 handles LTO natively.
        set(CMAKE_CXX_FLAGS_RELEASE "-O3 -march=native -flto=auto")
    endif()
endif()

# Add tests (placeholder)
enable_testing()
add_subdirectory(tests)

# Add examples (placeholder)
add_subdirectory(examples)

if(BUILD_SLICK_LOGGER_COLLECTOR)
    add_subdirectory(tools)
endif()

if(BUILD_SLICK_LOGGER_BENCHMARKS AND CMAKE_BUILD_TYPE MATCHES Release)
    add_subdirectory(benchmarks)
endif()

# Installation rules
install(DIRECTORY include/ DESTINATION include)

# Install CMake package configuration files for vcpkg
install(TARGETS slick-logger EXPORT slick-loggerTargets)

install(EXPORT slick-loggerTargets
    FILE slick-loggerTargets.cmake
    NAMESPACE slick::
    DESTINATION lib/cmake/slick-logger
)

include(CMakePackageConfigHelpers)

# Generate the config file
configure_package_config_file(
    ${CMAKE_CURRENT_SOURCE_DIR}/cmake/slick-loggerConfig.cmake.in
    ${CMAKE_CURRENT_BINARY_DIR}/slick-loggerConfig.cmake
    INSTALL_DESTINATION lib/cmake/slick-logger
)

# Generate version file
write_basic_package_version_file(
    ${CMAKE_CURRENT_BINARY_DIR}/slick-loggerConfigVersion.cmake
    VERSION ${PROJECT_VERSION}
    COMPATIBILITY SameMajorVersion
)

# Install config files
install(FILES
    ${CMAKE_CURRENT_BINARY_DIR}/slick-loggerConfig.cmake
    ${CMAKE_CURRENT_BINARY_DIR}/slick-loggerConfigVersion.cmake
    DESTINATION lib/cmake/slick-logger
)

message(STATUS "slick-logger: ${PROJECT_VERSION}")
